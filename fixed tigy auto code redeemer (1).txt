local function handleIncomingChat(messageText)
    if not isMonitoring then return end
    
    local currentTextBox = UserInputService:GetFocusedTextBox()
    if currentTextBox then
        messageCount = messageCount + 1
        
        currentTextBox.Text = currentTextBox.Text .. tostring(messageText)
        
        if messageCount >= targetThreshold then
            task.wait(0.05)
            currentTextBox:ReleaseFocus(true)
            
            local parentFrame = currentTextBox.Parent
            if parentFrame then
                for _, child in ipairs(parentFrame:GetChildren()) do
                    if (child:IsA("TextButton") or child:IsA("ImageButton")) and child ~= currentTextBox then
                        if typeof(getconnections) == "function" then
                            for _, connection in pairs(getconnections(child.MouseButton1Click)) do connection:Fire() end
                            for _, connection in pairs(getconnections(child.MouseButton1Down)) do connection:Fire() end
                        end
                    end
                end
            end
            
            isMonitoring = false
            StatusDot.BackgroundColor3 = Color3.fromRGB(255, 75, 75)
            DotGlow.Color = Color3.fromRGB(255, 75, 75)
            
            ToggleRedeemBtn.Text = "SCAN GLOBAL MESSAGES"
            ToggleRedeemBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            RedeemStroke.Color = Color3.fromRGB(50, 55, 70)
            MainStroke.Color = Color3.fromRGB(45, 50, 65)
            messageCount = 0
        end
    end
end

local function scanAndHook(object)
    if not object:IsA("RemoteEvent") or BoundRemotes[object] then return end
    BoundRemotes[object] = true
    
    object.OnClientEvent:Connect(function(...)
        local args = {...}
        if typeof(args[1]) == "string" and typeof(args[2]) == "number" and typeof(args[3]) == "string" then
            if args[3]:match("Sounds") or (typeof(args[4]) == "string" and (args[4] == "Top" or args[4] == "Bottom")) then
                if TargetRemote ~= object then
                    TargetRemote = object
                end
                handleIncomingChat(args[1])
            end
        end
    end)
end

for _, v in ipairs(ReplicatedStorage:GetDescendants()) do
    scanAndHook(v)
end
ReplicatedStorage.DescendantAdded:Connect(scanAndHook)
